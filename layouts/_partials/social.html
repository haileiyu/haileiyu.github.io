{{/* Open Graph tags, for the preview card a link gets when pasted into a chat app or social site. Called from head.html with dict "page" and "desc". The image is the page's `cover` (or else its first image), so photograph posts preview with a photograph and text posts with a plain text card. */}} {{ $p := .page }} {{ $img := "" }} {{ if $p.IsPage }} {{ with $p.Params.cover }}{{ $img = $p.Resources.GetMatch . }}{{ end }} {{ if not $img }} {{ with $p.Resources.ByType "image" }}{{ $img = index . 0 }}{{ end }} {{ end }} {{ end }} {{ with .desc }}{{ end }} {{ if and $p.IsPage (not $p.Date.IsZero) }} {{ end }} {{ with $img }} {{/* JPEG, because not every preview fetcher reads WebP. Never upscaled. */}} {{ $w := int (math.Min .Width 1200) }} {{ with .Resize (printf "%dx jpg q80" $w) }} {{ end }} {{ else }} {{ end }}
